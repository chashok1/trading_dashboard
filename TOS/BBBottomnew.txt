####   BridgeBands_PG — COMBINED study (TOP + BOTTOM bands), LEAN build v2
####   Same fitted values as TOS/BBTop.txt / BBBottom.txt (TASK_128–131,
####   calibrated vs hist_rr — docs/tos_rr_calibration.md).
####
#### ===========================  WHAT CHANGED vs v1  ==========================
####   1. ONE plot ("band") + a `side` input (TOP / BOTTOM). No more commenting
####      lines in and out. Works as a custom column (one plot) and on a chart
####      (add the study twice: once side=TOP, once side=BOTTOM).
####   2. Per-ticker k overrides gathered in ONE block, with an on/off switch
####      (useTickerKOverrides) and a manual override (kManual).
####   3. Magic number 15.87 replaced by Sqrt(252) (same value, self-explaining).
####   Numeric output is IDENTICAL to v1 with default inputs.
####
#### ===========================  USAGE  =======================================
#### CUSTOM COLUMN:  set `side` default below (TOP or BOTTOM) and save one
####                 column per side (a_bb_top / a_bb_bottom).
#### CHART:          add the study twice and pick `side` in each.
####
####   TOP    = topMid + topK*(1+topCIvHvWidth*(IV/HV-1))*sigma
####   BOTTOM = botMid - botKEff*(1+botCVolWidth*(RelVol-1))*(1+botCIvHvWidth*(IV/HV-1))
####                      *(1+botCVolLevel*(V/avg(V,m)-1))*sigma
####                   - botCVix*(vixTop/vixClose-1)*sigma
#### NOTE: BOTTOM's cVix/cVolLevel signs are empirical CV-gated results
#### (opposite naive intuition) — see docs before "fixing" them.
#### ===========================================================================

# ============================ [CONTROL] inputs ===============================
input side = {default BOTTOM, TOP};
input useTickerKOverrides = yes;  # no = ignore per-ticker k table, use topK/botK
input kManual = 0.0;              # > 0 forces k for this instance (chart testing)

# ============================ [SHARED] inputs ================================
input length = 10;          # EMA/StDev backbone window (n)
input volFastLen = 3;       # RelVol fast window (f)
input volSlowLen = 15;      # RelVol slow window (s)
input horizonDays = 3;      # IV-implied-move horizon (h; sqrt(h) scaling)
input sigmaWeight = 0.75;   # w: realized-vs-IV sigma blend weight
input vixWidthMult = 1.0;   # kv: VIX proxy-range width multiplier

# ============================ [TOP] inputs ===================================
input topMomFastLen = 12;    # momentum EMA fast window
input topMomSlowLen = 26;    # momentum EMA slow window
input topCTilt = 0.25;       # c_t: volume-confirmed midpoint tilt
input topCMom = 0.5;         # c_m: trend-momentum midpoint tilt
input topCIvHvWidth = -0.5;  # c_iv: width IV/HV multiplier
input topK = 1.3;            # width scale (default for all tickers)

# ============================ [BOTTOM] inputs ================================
input botMomFastLen = 5;     # momentum EMA fast window
input botMomSlowLen = 20;    # momentum EMA slow window
input botCTilt = 0.5;        # c_t: volume-confirmed midpoint tilt
input botCMom = 0.25;        # c_m: trend-momentum midpoint tilt
input botCVolWidth = -0.25;  # c_v: width RelVol multiplier (CV-accepted)
input botCIvHvWidth = -0.25; # c_iv: width IV/HV multiplier
input botK = 1.42;           # width scale (default for all tickers)
input botCVix = -0.5;        # c_vix: inverse-VIX coupling (CV-accepted; empirical sign)
input botVixLen = 20;        # VIX EMA/StDev window (BOTTOM fit)
input botCVolLevel = -0.25;  # c_lvl: vol-LEVEL width law (CV-accepted; empirical sign)
input botVolLevelLen = 20;   # vol-level rolling-average window

# ============================ [TICKER k TABLE] ===============================
# Why this exists: a watchlist column runs the SAME code on every row, and
# thinkScript has no lookup tables or per-row inputs. Checking the symbol is
# the only way to give a ticker its own calibrated k.
# To add a ticker: add one isXXX line + one line in each k chain below.
# (thinkScript cannot store strings in a def, so GetSymbol() is called per test.)
input topKVixOverride = 1.44;
input botKVixOverride = 1.28;
input botKOrclOverride = 1.22;
input botKNflxOverride = 2.04;

def isVix  = GetSymbol() == "VIX";
def isOrcl = GetSymbol() == "ORCL";
def isNflx = GetSymbol() == "NFLX";

def topKTable = if isVix then topKVixOverride
                else topK;
def botKTable = if isVix  then botKVixOverride
                else if isOrcl then botKOrclOverride
                else if isNflx then botKNflxOverride
                else botK;

def topKEff = if kManual > 0 then kManual else if useTickerKOverrides then topKTable else topK;
def botKEff = if kManual > 0 then kManual else if useTickerKOverrides then botKTable else botK;

# ============================ [SHARED] computation ===========================
# Sample stdev = built-in StDev (population) * bias correction.
def sampleFix = Sqrt(length / (length - 1.0));
def annFactor = Sqrt(252);

# --- price leg ---------------------------------------------------------------
def ema = ExpAverage(close, length);
def sdev = StDev(close, length) * sampleFix;

# --- volume leg: RelVol, clamped [0.5,2.0], neutral (1) when volume missing --
def volFast = Average(volume, volFastLen);
def volSlow = Average(volume, volSlowLen);
def relVolRaw = if IsNaN(volSlow) or volSlow == 0 then 1 else volFast / volSlow;
def relVol = Min(2.0, Max(0.5, relVolRaw));

# --- volatility leg: IV/HV ratio, clamped, neutral (1) when IV missing -------
def iv = imp_volatility();
def hvAnn = if close == 0 then 0 else sdev / close * annFactor;
def ivHvRaw = if IsNaN(iv) or hvAnn == 0 then 1 else iv / hvAnn;
def ivHv = Min(2.0, Max(0.5, ivHvRaw));

# --- sigma: realized/IV-implied blend (IV missing -> pure realized) ----------
def ivImplied = if IsNaN(iv) then 0 else close * iv / annFactor * Sqrt(horizonDays);
def wEff = if IsNaN(iv) then 1 else sigmaWeight;
def sigma = (wEff * sdev) + ((1 - wEff) * ivImplied);

# ============================ [TOP] band =====================================
def topEmaFastMom = ExpAverage(close, topMomFastLen);
def topEmaSlowMom = ExpAverage(close, topMomSlowLen);
def topMid = ema + (topCTilt * relVol * (close - ema)) + (topCMom * (topEmaFastMom - topEmaSlowMom));
def topIvHvWidthMult = Min(2.0, Max(0.5, 1 + (topCIvHvWidth * (ivHv - 1))));
def topBand = topMid + (topKEff * topIvHvWidthMult * sigma);

# ============================ [BOTTOM] band ==================================
def botEmaFastMom = ExpAverage(close, botMomFastLen);
def botEmaSlowMom = ExpAverage(close, botMomSlowLen);
def botMid = ema + (botCTilt * relVol * (close - ema)) + (botCMom * (botEmaFastMom - botEmaSlowMom));

# Vol-LEVEL width multiplier: own IV (fallback HV proxy) vs its rolling average.
def vLevelRaw = if IsNaN(iv) then hvAnn else iv;
def botVLevelAvg = Average(vLevelRaw, botVolLevelLen);
def botVLevelRatioRaw = if botVLevelAvg == 0 then 1 else vLevelRaw / botVLevelAvg;
def botVLevelRatio = Min(2.0, Max(0.5, botVLevelRatioRaw));
def botVLevelMult = Min(2.0, Max(0.5, 1 + (botCVolLevel * (botVLevelRatio - 1))));

# Inverse VIX coupling — BOTTOM uses VIX's UPSIDE room. Off on VIX itself.
def vixClose = close("VIX");
def botVixEma = ExpAverage(vixClose, botVixLen);
def botVixStd = StDev(vixClose, botVixLen) * Sqrt(botVixLen / (botVixLen - 1.0));
def botVixTop = botVixEma + (vixWidthMult * botVixStd);
def botVixTerm = if IsNaN(vixClose) or isVix or vixClose == 0 then 0 else (botVixTop / vixClose - 1);

def botVolWidthMult = Min(2.0, Max(0.5, 1 + (botCVolWidth * (relVol - 1))));
def botIvHvWidthMult = Min(2.0, Max(0.5, 1 + (botCIvHvWidth * (ivHv - 1))));
def botBand = botMid - (botKEff * botVolWidthMult * botIvHvWidthMult * botVLevelMult * sigma)
                     - (botCVix * botVixTerm * sigma);

# ============================ [OUTPUT] =======================================
plot band = if side == side.TOP then topBand else botBand;
